% Turkish dotted-İ fix: the class renders section titles, job titles, degree
% names, etc. via \scshape (small caps), which relies on the font's automatic
% lowercase-to-small-caps casing. Source Sans 3 doesn't carry a Turkish "locl"
% variant of that feature, so it silently drops the dot from "i" (e.g. "Lideri"
% renders as "LIDERI" instead of "LİDERİ"). There's no reliable font-level fix,
% so instead we disable the automatic casing here and type the affected strings
% in the content files as already-correct, manually-dotted Turkish upper case.
\renewcommand*{\headerpositionstyle}[1]{{\fontsize{7.6pt}{1em}\bodyfont\color{awesome} #1}}
\renewcommand*{\subsectionstyle}[1]{{\fontsize{12pt}{1em}\bodyfont\textcolor{text}{#1}}}
\renewcommand*{\entrypositionstyle}[1]{{\fontsize{8pt}{1em}\bodyfont\color{graytext} #1}}
